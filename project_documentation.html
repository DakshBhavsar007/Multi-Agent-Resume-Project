1.1 LLM Router Implementation (`backend/agents/llm.py`)

Python Code

Core load-balancer & failover client for Google Gemini & Groq LLM requests.

backend/agents/llm.py — RotateLLMClient Source Code Snippet Python
import google.generativeai as genai import os, time, logging from django.conf import settings logger = logging.getLogger(__name__) class RotateLLMClient: def __init__(self, agent_name="default"): self.agent_name = agent_name self.api_keys = getattr(settings, "GEMINI_API_KEYS", []) self.current_idx = 0 def generate_content(self, prompt: str, **kwargs) -> str: # Automatic failover across registered API key pool on 429 / 503 errors if not self.api_keys: self.api_keys = [os.getenv("GEMINI_API_KEY", "")] attempts = 0 max_attempts = len(self.api_keys) * 2 while attempts < max_attempts: key = self.api_keys[self.current_idx] try: genai.configure(api_key=key) model = genai.GenerativeModel('gemini-1.5-flash') res = model.generate_content(prompt, **kwargs) return res.text except Exception as e: logger.warning(f"Key {self.current_idx} failed for agent '{self.agent_name}': {e}. Rotating...") self.current_idx = (self.current_idx + 1) % len(self.api_keys) attempts += 1 time.sleep(0.5) raise RuntimeError("All Gemini API keys in pool exhausted.")

import pdfplumber, os, json from docx import Document from agents.llm import RotateLLMClient class ResumeParsingAgent: def __init__(self): self.client = RotateLLMClient(agent_name="parsing_agent") async def parse(self, file_path: str, file_type: str) -> dict: if file_type == "pdf": text = self._extract_pdf_text(file_path) elif file_type in ["docx", "doc"]: text = self._extract_docx_text(file_path) else: with open(file_path, 'r', errors='ignore') as f: text = f.read() prompt = f""" Extract candidate profile JSON from this resume text: {text[:4000]} Return pure JSON matching: {{ "name": str, "email": str|null, "phone": str|null, "location": str|null, "skills": [{"skill": str, "years": float|null}], "total_experience_years": float, "experience": [{"company": str, "role": str, "responsibilities": [str]}] }} """ raw_response = self.client.generate_content(prompt) return json.loads(raw_response.strip("`json\n"))

1.3 Semantic Matching Agent (`backend/agents/matching_agent.py`)

Python Code
backend/agents/matching_agent.py — Vector Similarity Implementation Python
import numpy as np from sklearn.metrics.pairwise import cosine_similarity from agents.embeddings import get_embedding_model class SemanticMatchingAgent: async def match(self, candidate: dict, criteria: dict) -> dict: weights = criteria.get("weights", {"skills": 0.5, "experience": 0.3, "location": 0.2}) required = criteria.get("required_skills", []) candidate_skills = [s.get("canonical_skill", str(s)) for s in candidate.get("normalized_skills", [])] model = get_embedding_model() req_embeddings = model.encode(required) cand_embeddings = model.encode(candidate_skills) # Compute Cosine Similarity Matrix sim_matrix = cosine_similarity(cand_embeddings, req_embeddings) matched = [] for i, req in enumerate(required): sims = sim_matrix[:, i] if float(np.max(sims)) > 0.72: matched.append(req) skill_score = (len(matched) / len(required)) * 100 if required else 70.0 final_score = (skill_score * weights["skills"]) + 30.0 return {"match_score": round(final_score, 1), "matched_skills": matched}
